set(STANDELONE_MODULES TRUE PARENT_SCOPE)

set(NYAANWORK_ASSET FALSE CACHE BOOL "Enable Asset module" FORCE)
set(NYAANWORK_ASSET_IMAGE FALSE CACHE BOOL "Enable Asset.Image submodule" FORCE)
set(NYAANWORK_ASSET_TILE FALSE CACHE BOOL "Enable Asset.Tile submodule" FORCE)

if (CONFIG_NYAANWORK_ASSET)
	set(NYAANWORK_ASSET TRUE CACHE BOOL "Enable Asset module" FORCE)

	if (CONFIG_NYAANWORK_ASSET_IMAGE)
		list_directory("${CMAKE_CURRENT_SOURCE_DIR}/Image" cpp MODULES_IMAGE)
		set(NYAANWORK_ASSET_IMAGE TRUE
			CACHE BOOL "Enable Asset.Image submodule" FORCE)
	endif()

	if (CONFIG_NYAANWORK_ASSET_TILE)
		list_directory("${CMAKE_CURRENT_SOURCE_DIR}/Tile" cpp MODULES_TILE)
		set(NYAANWORK_ASSET_TILE TRUE
			CACHE BOOL "Enable Asset.Tile submodule" FORCE)
	endif()
endif()

set(MODULES ${MODULES_IMAGE} ${MODULES_TILE}
	"${CMAKE_CURRENT_SOURCE_DIR}/Exception.cpp" PARENT_SCOPE)
