set(NYAANWORK_HIGH_OPENGL FALSE CACHE BOOL "Enable HOGL module" FORCE)

set(NYAANWORK_HIGH_OPENGL_CORE330 FALSE CACHE BOOL
	"HOGL: Enable 330 Core context" FORCE)
set(NYAANWORK_HIGH_OPENGL_CORE330_DYNAMIC_LIMITS BOOL CACHE INTERNAL
	"HOGL.Core330: Dynamic limits" FORCE)
set(NYAANWORK_HIGH_OPENGL_CORE330_TEXTURE_SLOTS 1 CACHE INTERNAL
	"HOGL.Core330: Texture slots limit" FORCE)
set(NYAANWORK_HIGH_OPENGL_CORE330_TEXTURE_SIZE 1 CACHE INTERNAL
	"HOGL.Core330: Texture size limit" FORCE)
set(NYAANWORK_HIGH_OPENGL_CORE330_UNIFROM_BUFFER_BINDINGS 1 CACHE INTERNAL
	"HOGL.Core330: Unifrom buffer bindings limit" FORCE)
set(NYAANWORK_HIGH_OPENGL_CORE330_UNIFROM_BLOCK_SIZE 1 CACHE INTERNAL
	"HOGL.Core330: Unifrom block size limit" FORCE)

set(modules "${CMAKE_CURRENT_SOURCE_DIR}/Exception.cpp")

if (CONFIG_NYAANWORK_HIGH_OPENGL)
	set(NYAANWORK_HIGH_OPENGL TRUE CACHE BOOL "Enable HOGL module" FORCE)

	if (CONFIG_NYAANWORK_HIGH_OPENGL_CORE330)
		set(NYAANWORK_HIGH_OPENGL_CORE330 TRUE CACHE BOOL
			"HOGL: 330 Core context" FORCE)

		list_directory("${CMAKE_CURRENT_SOURCE_DIR}/Core330" cpp MODULES_CORE330)
		list(APPEND modules ${MODULES_CORE330})

		set(NYAANWORK_HIGH_OPENGL_CORE330_DYNAMIC_LIMITS
			${CONFIG_NYAANWORK_HIGH_OPENGL_CORE330_DYNAMIC_LIMITS}
			CACHE BOOL "HOGL.Core330: Dynamic limits" FORCE)

		set(NYAANWORK_HIGH_OPENGL_CORE330_TEXTURE_SLOTS
			${CONFIG_NYAANWORK_HIGH_OPENGL_CORE330_TEXTURE_SLOTS}
			CACHE INTERNAL "HOGL.Core330: Texture slots limit" FORCE)
		set(NYAANWORK_HIGH_OPENGL_CORE330_TEXTURE_SIZE
			${CONFIG_NYAANWORK_HIGH_OPENGL_CORE330_TEXTURE_SIZE}
			CACHE INTERNAL "HOGL.Core330: Texture size limit" FORCE)
		set(NYAANWORK_HIGH_OPENGL_CORE330_UNIFROM_BUFFER_BINDINGS
			${CONFIG_NYAANWORK_HIGH_OPENGL_CORE330_UNIFROM_BUFFER_BINDINGS}
			CACHE INTERNAL "HOGL.Core330: Unifrom buffer bindings limit" FORCE)
		set(NYAANWORK_HIGH_OPENGL_CORE330_UNIFROM_BLOCK_SIZE
			${CONFIG_NYAANWORK_HIGH_OPENGL_CORE330_UNIFROM_BLOCK_SIZE}
			CACHE INTERNAL "HOGL.Core330: Unifrom block size limit" FORCE)
	endif()
endif()
